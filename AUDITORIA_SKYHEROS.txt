package br.com.skyheros.ability;

import com.mojang.brigadier.arguments.StringArgumentType;
import net.minecraft.commands.Commands;
import net.minecraft.network.chat.Component;
import net.minecraft.server.level.ServerPlayer;
import net.minecraftforge.event.RegisterCommandsEvent;

public final class AbilityCommands {
    private AbilityCommands() {}

    public static void register(RegisterCommandsEvent event) {
        event.getDispatcher().register(
            Commands.literal("skyheros")
                .then(Commands.literal("ability")
                    .then(Commands.argument("id", StringArgumentType.word()).suggests(br.com.skyheros.command.CatalogSuggestions::abilities)
                        .executes(context -> {
                            if (!(context.getSource().getEntity() instanceof ServerPlayer player)) {
                                context.getSource().sendFailure(Component.literal(
                                    "Este comando deve ser usado por um jogador."
                                ));
                                return 0;
                            }
                            return AbilityRuntime.activate(
                                player,
                                StringArgumentType.getString(context, "id")
                            );
                        })
                    )
                )
                .then(Commands.literal("list")
                    .executes(context -> {
                        String ids = AbilityRegistry.getAll().stream()
                            .map(AbilityDefinition::id)
                            .sorted()
                            .limit(40)
                            .reduce((a, b) -> a + ", " + b)
                            .orElse("nenhuma");

                        context.getSource().sendSuccess(() -> Component.literal(
                            "Catalogo: " + AbilityAudit.totalCount()
                                + " habilidades. Primeiras 40 por ID: " + ids
                        ), false);
                        return 1;
                    })
                )
                .then(Commands.literal("catalog")
                    .executes(context -> {
                        context.getSource().sendSuccess(() -> Component.literal(
                            "Skyheros | Total: " + AbilityAudit.totalCount()
                                + " | Com executor: " + AbilityAudit.implementedCount()
                                + " | Aguardando executor: " + AbilityAudit.pendingCount()
                        ), false);
                        return 1;
                    })
                )
                .then(Commands.literal("validate")
                    .executes(context -> {
                        int total = AbilityAudit.totalCount();
                        long implemented = AbilityAudit.implementedCount();
                        long pending = AbilityAudit.pendingCount();
                        boolean correct = total == 1000 && implemented + pending == total;

                        context.getSource().sendSuccess(() -> Component.literal(
                            (correct ? "[OK] " : "[ATENCAO] ")
                                + "Catalogo: " + total + "/1000"
                                + " | Executores implementados: " + implemented
                                + " | Pendentes: " + pending
                        ), false);
                        return correct ? 1 : 0;
                    })
                )
        );
    }
}

package br.com.skyheros.ability;

import java.util.Collection;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Optional;

public final class AbilityRegistry {
    private static final Map<String, AbilityDefinition> ABILITIES = new LinkedHashMap<>();

    private AbilityRegistry() {}

    public static synchronized void register(AbilityDefinition ability) {
        if (ABILITIES.putIfAbsent(ability.id(), ability) != null) {
            throw new IllegalArgumentException("Habilidade duplicada: " + ability.id());
        }
    }

    public static synchronized Optional<AbilityDefinition> get(String id) {
        return Optional.ofNullable(ABILITIES.get(id));
    }

    public static synchronized Collection<AbilityDefinition> getAll() {
        return Collections.unmodifiableList(java.util.List.copyOf(ABILITIES.values()));
    }

    public static synchronized int size() {
        return ABILITIES.size();
    }
}

package br.com.skyheros.ability;

import net.minecraft.commands.CommandSourceStack;
import net.minecraft.world.entity.Entity;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.effect.MobEffectInstance;
import net.minecraft.world.effect.MobEffects;
import net.minecraft.world.phys.AABB;
import net.minecraft.world.phys.BlockHitResult;
import net.minecraft.world.phys.HitResult;
import net.minecraft.world.phys.Vec3;
import net.minecraft.world.level.ClipContext;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.network.chat.Component;
import net.minecraft.world.entity.projectile.ProjectileUtil;

import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;

public final class AbilityRuntime {
    private static final Map<String, Long> COOLDOWNS = new ConcurrentHashMap<>();

    private AbilityRuntime() {}

    public static int activate(ServerPlayer player, String id) {
        AbilityDefinition ability = AbilityRegistry.get(id).orElse(null);
        if (ability == null) {
            player.sendSystemMessage(Component.literal("Habilidade desconhecida: " + id));
            return 0;
        }

        long now = player.level().getGameTime();
        String key = player.getUUID() + ":" + ability.id();
        long until = COOLDOWNS.getOrDefault(key, 0L);

        if (now < until) {
            player.sendSystemMessage(Component.literal(
                "Habilidade em recarga: " + ((until - now + 19) / 20) + " s"
            ));
            return 0;
        }

        boolean success = execute(player, ability.behavior());
        if (!success) {
            player.sendSystemMessage(Component.literal(
                "Nao foi possivel usar " + ability.name() + ". Mire em um alvo ou bloco valido."
            ));
            return 0;
        }

        COOLDOWNS.put(key, now + ability.cooldownTicks());
        player.sendSystemMessage(Component.literal("Habilidade: " + ability.name()));
        return 1;
    }

    private static boolean execute(ServerPlayer player, String behavior) {
        Vec3 eye = player.getEyePosition();
        Vec3 look = player.getLookAngle();
        Vec3 end = eye.add(look.scale(24.0D));

        switch (behavior) {
            case "WEB_SWING": {
                BlockHitResult hit = player.level().clip(new ClipContext(
                    eye, end, ClipContext.Block.OUTLINE,
                    ClipContext.Fluid.NONE, player
                ));
                if (hit.getType() == HitResult.Type.MISS) return false;

                Vec3 anchor = Vec3.atCenterOf(hit.getBlockPos());
                Vec3 toward = anchor.subtract(player.position());
                if (toward.lengthSqr() < 1.0D) return false;

                Vec3 impulse = toward.normalize().scale(0.72D).add(0.0D, 0.28D, 0.0D);
                player.setDeltaMovement(player.getDeltaMovement().add(impulse));
                player.fallDistance = 0.0F;
                player.hurtMarked = true;
                return true;
            }

            case "WEB_SHOT":
            case "WEB_PULL":
            case "WEB_BIND": {
                Entity target = findTarget(player, eye, end, look);
                if (!(target instanceof LivingEntity living)) return false;

                if ("WEB_SHOT".equals(behavior)) {
                    living.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 50, 1));
                    return true;
                }

                if ("WEB_PULL".equals(behavior)) {
                    Vec3 pull = player.position().subtract(living.position());
                    if (pull.lengthSqr() < 0.01D) return false;
                    living.setDeltaMovement(living.getDeltaMovement().add(pull.normalize().scale(0.8D)));
                    living.hurtMarked = true;
                    return true;
                }

                living.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 100, 4));
                living.addEffect(new MobEffectInstance(MobEffects.JUMP, 100, 128));
                return true;
            }

            case "SPIDER_SENSE":
                player.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SPEED, 160, 1));
                player.addEffect(new MobEffectInstance(MobEffects.NIGHT_VISION, 160, 0));
                player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 80, 0));
                return true;

            case "ENHANCED_LEAP":
                player.setDeltaMovement(player.getDeltaMovement()
                    .add(look.x * 0.65D, 0.85D, look.z * 0.65D));
                player.fallDistance = 0.0F;
                player.hurtMarked = true;
                return true;

            case "SUPER_STRENGTH":
                player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_BOOST, 160, 1));
                player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 80, 0));
                return true;

            default:
                player.sendSystemMessage(Component.literal(
                    "Esta habilidade esta cadastrada, mas seu executor ainda sera implementado: " + behavior
                ));
                return false;
        }
    }

    private static Entity findTarget(ServerPlayer player, Vec3 start, Vec3 end, Vec3 look) {
        AABB box = player.getBoundingBox()
            .expandTowards(look.scale(24.0D))
            .inflate(1.5D);

        var result = ProjectileUtil.getEntityHitResult(
            player,
            start,
            end,
            box,
            entity -> entity != player
                && entity.isPickable()
                && !entity.isSpectator()
                && entity instanceof LivingEntity,
            24.0D * 24.0D
        );

        return result == null ? null : result.getEntity();
    }
}

package br.com.skyheros.power;

import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.effect.MobEffectInstance;
import net.minecraft.world.effect.MobEffects;
import net.minecraft.world.entity.EntityType;
import net.minecraft.world.entity.Mob;
import net.minecraft.world.entity.LivingEntity;
import net.minecraft.world.entity.animal.Wolf;
import net.minecraft.world.phys.Vec3;
import java.util.Comparator;
import java.util.List;
import java.util.Locale;

public final class CombatProgressionRuntime {
    private CombatProgressionRuntime() {}

    private static boolean hasAny(String text, String... terms) {
        for (String term : terms) if (text.contains(term)) return true;
        return false;
    }

    public static String execute(ServerPlayer player, PowerDefinition power) {
        String id = power.getId().toLowerCase(Locale.ROOT);
        if (!id.startsWith("combo_")) return null;

        String key = (id + " " + power.getName() + " "
                + power.getTheme() + " " + power.getMechanic() + " "
                + power.getBehavior()).toLowerCase(Locale.ROOT);
        String mechanic = power.getMechanic().toLowerCase(Locale.ROOT);
        double strength = Math.max(1.0, Math.min(8.0, power.getMagnitude()));
        Vec3 look = player.getLookAngle().normalize();

        if (mechanic.equals("transformation")) {
            player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_BOOST, 240, 1));
            player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 240, 1));
            return power.getName() + " ativado por 12 segundos.";
        }

        if (mechanic.equals("support")) {
            List<ServerPlayer> allies = player.level().getEntitiesOfClass(
                    ServerPlayer.class, player.getBoundingBox().inflate(8.0),
                    entity -> entity.isAlive());
            for (ServerPlayer ally : allies) {
                if (hasAny(key, "regeneration", "revitalizing", "cleanse", "sacred")) {
                    ally.heal(4.0F);
                    ally.addEffect(new MobEffectInstance(MobEffects.REGENERATION, 100, 0));
                }
                if (hasAny(key, "shield", "guardian", "barrier", "sacred")) {
                    ally.addEffect(new MobEffectInstance(MobEffects.ABSORPTION, 200, 1));
                    ally.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 160, 0));
                }
                if (hasAny(key, "speed", "rally", "warcry", "coordination", "focus")) {
                    ally.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SPEED, 200, 1));
                    ally.addEffect(new MobEffectInstance(MobEffects.DAMAGE_BOOST, 160, 0));
                }
                if (hasAny(key, "cleanse")) {
                    ally.removeEffect(MobEffects.POISON);
                    ally.removeEffect(MobEffects.WITHER);
                    ally.removeEffect(MobEffects.WEAKNESS);
                    ally.removeEffect(MobEffects.BLINDNESS);
                    ally.removeEffect(MobEffects.MOVEMENT_SLOWDOWN);
                }
            }
            return power.getName() + " aplicado a " + allies.size() + " jogador(es) próximo(s).";
        }

        if (mechanic.equals("defense")) {
            player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 240, 1));
            player.addEffect(new MobEffectInstance(MobEffects.ABSORPTION, 240, 1));
            if (hasAny(key, "fireproof", "elemental_ward")) {
                player.addEffect(new MobEffectInstance(MobEffects.FIRE_RESISTANCE, 240, 0));
            }
            if (hasAny(key, "frost")) {
                player.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SPEED, 160, 0));
            }
            if (hasAny(key, "regeneration_matrix", "auto_recovery", "adaptive_armor")) {
                player.addEffect(new MobEffectInstance(MobEffects.REGENERATION, 160, 0));
            }
            return power.getName() + ": defesa reforçada por 12 segundos.";
        }

        if (mechanic.equals("passive")) {
            if (hasAny(key, "toxin_purge")) {
                player.removeEffect(MobEffects.POISON);
                player.removeEffect(MobEffects.WITHER);
                player.addEffect(new MobEffectInstance(MobEffects.REGENERATION, 100, 0));
            } else {
                player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_RESISTANCE, 200, 0));
                if (hasAny(key, "shadow", "stealth")) {
                    player.addEffect(new MobEffectInstance(MobEffects.INVISIBILITY, 160, 0));
                }
                if (hasAny(key, "recovery")) {
                    player.addEffect(new MobEffectInstance(MobEffects.REGENERATION, 200, 0));
                }
            }
            return power.getName() + " ativado temporariamente.";
        }

        if (mechanic.equals("movement")) {
            if (hasAny(key, "blink", "phase_dash", "spatial_snap")) {
                double distance = key.contains("spatial_snap") ? 10.0 : 6.0;
                Vec3 destination = player.position().add(look.scale(distance));
                player.teleportTo(destination.x, destination.y, destination.z);
                player.fallDistance = 0;
                return power.getName() + ": deslocamento executado.";
            }
            if (hasAny(key, "speed", "afterimage", "sonic_dash")) {
                player.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SPEED, 160, 2));
            }
            if (hasAny(key, "leap", "skybound", "drift", "gravity_sling")) {
                player.addEffect(new MobEffectInstance(MobEffects.SLOW_FALLING, 200, 0));
                player.setDeltaMovement(player.getDeltaMovement().add(
                        look.x * strength * 0.45, Math.max(0.5, strength * 0.18),
                        look.z * strength * 0.45));
                player.hurtMarked = true;
            }
            return power.getName() + ": movimento especial ativado.";
        }

        if (mechanic.equals("utility")) {
            if (hasAny(key, "energy_recharge")) {
                String restored = PowerProgressionRuntime.restoreEnergy(player, 25);
                return "Energia restaurada. " + restored;
            }
            if (hasAny(key, "power_surge")) {
                player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_BOOST, 200, 1));
                player.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SPEED, 200, 1));
                return "Surto de poder ativado.";
            }
            player.addEffect(new MobEffectInstance(MobEffects.NIGHT_VISION, 300, 0));
            List<LivingEntity> detected = player.level().getEntitiesOfClass(
                    LivingEntity.class, player.getBoundingBox().inflate(16.0),
                    entity -> entity != player && entity.isAlive());
            for (LivingEntity entity : detected) {
                entity.addEffect(new MobEffectInstance(MobEffects.GLOWING, 120, 0));
            }
            return "Varredura concluída: " + detected.size() + " entidade(s) detectada(s).";
        }

        if (mechanic.equals("environment")) {
            List<LivingEntity> nearby = player.level().getEntitiesOfClass(
                    LivingEntity.class, player.getBoundingBox().inflate(8.0),
                    entity -> entity != player && entity.isAlive());

            if (hasAny(key, "stormfront")) {
                player.serverLevel().setWeatherParameters(0, 600, true, true);
            } else if (hasAny(key, "frost_domain")) {
                for (LivingEntity entity : nearby) {
                    entity.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 120, 3));
                    entity.addEffect(new MobEffectInstance(MobEffects.WEAKNESS, 100, 0));
                }
            } else if (hasAny(key, "seismic")) {
                for (LivingEntity entity : nearby) {
                    entity.hurt(player.damageSources().playerAttack(player), 4.0F);
                    Vec3 away = entity.position().subtract(player.position()).normalize();
                    entity.push(away.x, 0.45, away.z);
                    entity.hurtMarked = true;
                }
            } else if (hasAny(key, "gravitational")) {
                for (LivingEntity entity : nearby) {
                    entity.addEffect(new MobEffectInstance(MobEffects.LEVITATION, 50, 0));
                    entity.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 100, 1));
                }
            } else {
                for (LivingEntity entity : nearby) {
                    entity.setSecondsOnFire(4);
                    entity.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 80, 1));
                }
            }
            return power.getName() + " afetou a área próxima (" + nearby.size() + " criatura(s)).";
        }

        if (mechanic.equals("summon")) {
            EntityType<? extends Mob> type = EntityType.WOLF;
            if (hasAny(key, "iron_sentinel")) type = EntityType.IRON_GOLEM;
            else if (hasAny(key, "skeleton_guard")) type = EntityType.SKELETON;
            else if (hasAny(key, "swarm")) type = EntityType.BEE;

            int count = hasAny(key, "pack", "swarm", "hounds") ? 3 : 1;
            int spawned = 0;
            for (int i = 0; i < count; i++) {
                Mob ally = type.create(player.level());
                if (ally == null) continue;
                double angle = 2.0 * Math.PI * i / count;
                ally.moveTo(player.getX() + Math.cos(angle) * 2.0,
                        player.getY(), player.getZ() + Math.sin(angle) * 2.0,
                        player.getYRot(), 0.0F);
                if (ally instanceof Wolf wolf) {
                    wolf.setOwnerUUID(player.getUUID());
                    wolf.setTame(true);
                }
                if (player.level().addFreshEntity(ally)) spawned++;
            }
            return spawned + " aliado(s) invocado(s).";
        }

        LivingEntity target = findTarget(player, 16.0);

        if (mechanic.equals("control")) {
            if (target == null) return "Nenhum alvo na mira a até 16 blocos.";
            if (hasAny(key, "gravity", "vortex", "telekinetic")) {
                target.addEffect(new MobEffectInstance(MobEffects.LEVITATION, 60, 1));
            }
            target.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SLOWDOWN, 100, 3));
            target.addEffect(new MobEffectInstance(MobEffects.WEAKNESS, 80, 0));
            if (hasAny(key, "repulsion")) {
                Vec3 away = target.position().subtract(player.position()).normalize();
                target.push(away.x * 1.5, 0.5, away.z * 1.5);
                target.hurtMarked = true;
            }
            return power.getName() + " controlou " + target.getName().getString() + ".";
        }

        if (mechanic.equals("attack")) {
            if (target == null) return "Nenhum alvo na mira a até 16 blocos.";

            float damage = (float)Math.max(3.0, Math.min(20.0, strength * 2.5));
            if (hasAny(key, "ultimate", "finisher", "annihilation", "worldbreaker",
                    "hypernova", "collapse", "singularity")) {

                    + " e causou " + damage + " de dano.";
        }

        String universalResult = UniversalPowerRuntime.execute(player, power);
        if (universalResult != null) return universalResult;
        return "Combo sem executor universal.";
    }

    private static LivingEntity findTarget(ServerPlayer player, double range) {
        Vec3 look = player.getLookAngle().normalize();
        List<LivingEntity> entities = player.level().getEntitiesOfClass(
                LivingEntity.class, player.getBoundingBox().inflate(range),
                entity -> entity != player && entity.isAlive());
        return entities.stream().filter(entity -> {
            Vec3 diff = entity.getEyePosition().subtract(player.getEyePosition());
            double distance = diff.length();
            return distance > 0.1 && distance <= range
                    && look.dot(diff.normalize()) > 0.65
                    && player.hasLineOfSight(entity);
        }).min(Comparator.comparingDouble(player::distanceTo)).orElse(null);
    }

package br.com.skyheros.power;

import net.minecraft.nbt.CompoundTag;
import net.minecraft.server.level.ServerPlayer;
import net.minecraft.world.effect.MobEffectInstance;
import net.minecraft.world.effect.MobEffects;

public final class PowerProgressionRuntime {
    private static final String KEY = "skyheros_progression";
    private static final int MAX_ENERGY = 100;

    private PowerProgressionRuntime() {}

    private static CompoundTag state(ServerPlayer player) {
        CompoundTag root = player.getPersistentData();
        if (!root.contains(KEY)) root.put(KEY, new CompoundTag());
        return root.getCompound(KEY);
    }

    private static void save(ServerPlayer player, CompoundTag data) {
        player.getPersistentData().put(KEY, data);
    }

    private static int refresh(ServerPlayer player, CompoundTag data) {
        long now = player.level().getGameTime();

        if (!data.contains("energy")) data.putInt("energy", MAX_ENERGY);
        if (!data.contains("last_energy_tick")) data.putLong("last_energy_tick", now);

        long last = data.getLong("last_energy_tick");
        long elapsed = Math.max(0L, now - last);
        int regenerated = (int)Math.min(MAX_ENERGY, elapsed / 20L);
        int energy = Math.min(MAX_ENERGY, data.getInt("energy") + regenerated);

        data.putInt("energy", energy);
        if (energy >= MAX_ENERGY) data.putLong("last_energy_tick", now);
        else data.putLong("last_energy_tick", last + regenerated * 20L);

        save(player, data);
        return energy;
    }

    public static String checkEnergy(ServerPlayer player, PowerDefinition power) {
        CompoundTag data = state(player);
        int energy = refresh(player, data);
        int cost = Math.max(0, Math.min(MAX_ENERGY, (int)Math.ceil(power.getEnergyCost())));

        if (energy < cost) {
            return "Energia insuficiente: " + energy + "/" + MAX_ENERGY
                    + ". Este poder custa " + cost + ". Regenera 1 ponto por segundo.";
        }
        return null;
    }

    public static String restoreEnergy(ServerPlayer player, int amount) {
        CompoundTag data = state(player);
        int energy = refresh(player, data);
        energy = Math.min(MAX_ENERGY, energy + Math.max(0, amount));
        data.putInt("energy", energy);
        data.putLong("last_energy_tick", player.level().getGameTime());
        save(player, data);
        return "Energia: " + energy + "/" + MAX_ENERGY + ".";
    }

    public static String recordSuccessfulUse(ServerPlayer player, PowerDefinition power) {
        CompoundTag data = state(player);
        int energy = refresh(player, data);
        int cost = Math.max(0, Math.min(MAX_ENERGY, (int)Math.ceil(power.getEnergyCost())));
        energy = Math.max(0, energy - cost);
        data.putInt("energy", energy);
        data.putLong("last_energy_tick", player.level().getGameTime());

        String powerKey = power.getId().replace(':', '_').replace('.', '_');
        CompoundTag mastery = data.getCompound("mastery");
        int uses = mastery.getInt(powerKey) + 1;
        mastery.putInt(powerKey, uses);
        data.put("mastery", mastery);

        int level = Math.min(10, 1 + uses / 10);
        String message = "Energia " + energy + "/" + MAX_ENERGY
                + " | Maestria " + level + " (" + uses + " usos)";

        if (uses % 10 == 0) {
            player.addEffect(new MobEffectInstance(MobEffects.ABSORPTION, 100, 1));
            message += " | Marco de maestria alcançado!";
        }

        long now = player.level().getGameTime();
        long previous = data.getLong("last_combo_tick");
        int combo = (previous > 0 && now >= previous && now - previous <= 100)
                ? data.getInt("combo_count") + 1 : 1;

        data.putLong("last_combo_tick", now);
        if (combo >= 3) {
            player.addEffect(new MobEffectInstance(MobEffects.MOVEMENT_SPEED, 80, 1));
            player.addEffect(new MobEffectInstance(MobEffects.DAMAGE_BOOST, 80, 0));
            data.putInt("combo_count", 0);
            message += " | COMBO x3! Bônus de velocidade e dano por 4 segundos.";
        } else {
            data.putInt("combo_count", combo);
            message += " | Combo " + combo + "/3";
        }

        save(player, data);
        return message;
    }
}

package br.com.skyheros.power;

import java.util.ArrayList;
import java.util.Collection;
import java.util.Collections;
import java.util.LinkedHashMap;
import java.util.Locale;
import java.util.Map;
import java.util.Optional;

public final class PowerRegistry {
    private static final Map<String, PowerDefinition> POWERS =
            new LinkedHashMap<>();

    private PowerRegistry() {}

    public static synchronized void register(PowerDefinition power) {
        if (power == null) {
            throw new IllegalArgumentException("Poder nulo");
        }

        String id = power.getId().toLowerCase(Locale.ROOT);

        if (POWERS.containsKey(id)) {
            throw new IllegalArgumentException("ID de poder duplicado: " + id);
        }

        POWERS.put(id, power);
    }

    public static synchronized Optional<PowerDefinition> get(String id) {
        if (id == null) return Optional.empty();
        return Optional.ofNullable(POWERS.get(id.toLowerCase(Locale.ROOT)));
    }

    public static synchronized Collection<PowerDefinition> getAll() {
        return Collections.unmodifiableList(
                new ArrayList<>(POWERS.values()));
    }

    public static synchronized int size() {
        return POWERS.size();
    }
}

    private static LivingEntity findTarget(ServerPlayer player, double range) {
        Vec3 look = player.getLookAngle().normalize();
        List<LivingEntity> entities = player.level().getEntitiesOfClass(
                LivingEntity.class,
                player.getBoundingBox().inflate(range),
                entity -> entity != player && entity.isAlive()
        );

        return entities.stream()
                .filter(entity -> {
                    Vec3 difference = entity.getEyePosition()
                            .subtract(player.getEyePosition());
                    double distance = difference.length();
                    return distance > 0.1 && distance <= range
                            && look.dot(difference.normalize()) > 0.65
                            && player.hasLineOfSight(entity);
                })
                .min(Comparator.comparingDouble(player::distanceTo))
                .orElse(null);
    }

    private static LivingEntity getTarget(ServerPlayer player, double range) {
        HitResult hit = player.pick(range, 1.0F, false);

        if (hit.getType() != HitResult.Type.ENTITY) return null;
        if (!(hit instanceof EntityHitResult entityHit)) return null;

        if (!(entityHit.getEntity() instanceof LivingEntity target)) return null;
        if (target == player || !target.isAlive()) return null;

        return target;

    private static String lower(String value) {

    private static double clamp(double value, double min, double max) {
        return Math.max(min, Math.min(max, value));
